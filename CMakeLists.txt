# Keep JUCE headers, definitions and system dependencies on its interface targets,
# but compile their implementation sources once into our shared static runtime.
# juce_audio_plugin_client otherwise pulls all GUI/processor sources into each
# of the 50 shared-code targets through its transitive module dependencies.
foreach(module juce_core juce_events juce_data_structures juce_graphics juce_gui_basics juce_gui_extra
 juce_audio_basics juce_audio_devices juce_audio_formats juce_audio_processors juce_audio_processors_headless
 juce_audio_utils juce_dsp juce_cryptography)
 get_target_property(module_sources ${module} INTERFACE_SOURCES)
 if(module_sources)
  target_sources(hg_runtime PRIVATE ${module_sources})
  set_property(TARGET ${module} PROPERTY INTERFACE_SOURCES "")
 endif()
endforeach()

function(hg_add_product name index code category)
 set(formats VST3)
 if(HG_BUILD_STANDALONE AND name STREQUAL "RIFT")
  list(APPEND formats Standalone)
 endif()
 set(product "Hungry Ghost ${name}")
 set(version "0.1.0")
 string(TOLOWER "${name}" slug)
 if(name STREQUAL "REVERB")
  set(product "Hungry Ghost")
  set(version "0.3.0")
 elseif(name STREQUAL "FERAL")
  set(version "0.2.0")
 endif()
 juce_add_plugin(${name} VERSION ${version} COMPANY_NAME "Hungry Ghost Audio" BUNDLE_ID "audio.hungryghost.${slug}"
  PLUGIN_MANUFACTURER_CODE Aftr PLUGIN_CODE ${code} FORMATS ${formats} PRODUCT_NAME "${product}"
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
  EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx)
 if(index LESS 2)
  target_sources(${name} PRIVATE Source/OriginalEntry.cpp)
 else()
  target_sources(${name} PRIVATE Source/PluginEntry.cpp)
 endif()
 target_compile_definitions(${name} PRIVATE HG_PRODUCT_INDEX=${index})
 target_compile_definitions(${name} PUBLIC $<TARGET_PROPERTY:hg_runtime,COMPILE_DEFINITIONS>)
 target_include_directories(${name} PUBLIC $<TARGET_PROPERTY:hg_runtime,INCLUDE_DIRECTORIES>)
 target_link_libraries(${name} PRIVATE hg_runtime)
 list(APPEND HG_PRODUCT_TARGETS ${name}_VST3)
 set(HG_PRODUCT_TARGETS "${HG_PRODUCT_TARGETS}" PARENT_SCOPE)
endfunction()
